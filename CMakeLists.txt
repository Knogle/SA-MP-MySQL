cmake_minimum_required(VERSION 3.19)
project(mysql)

set(MYSQL_PLUGIN_VERSION "R42")

option (BUILD_TESTS "Build tests" OFF)
option (FORCE_32_BIT "Force 32-bit compilation on UNIX" ON)
if(WIN32)
	set(_DEFAULT_BUILD_BUNDLED_MARIADB_CONNECTOR OFF)
else()
	set(_DEFAULT_BUILD_BUNDLED_MARIADB_CONNECTOR ON)
endif()
option(BUILD_BUNDLED_MARIADB_CONNECTOR "Build bundled MariaDB Connector/C submodule from source." ${_DEFAULT_BUILD_BUNDLED_MARIADB_CONNECTOR})

list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/libs/cmake")

if(UNIX)
	#force 32bit compilation when requested
	if(FORCE_32_BIT)
		set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -m32")
		set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -m32")
		set_property(GLOBAL PROPERTY FIND_LIBRARY_USE_LIB64_PATHS OFF)
	endif()
endif()

set(_BUNDLED_MARIADB_ROOT "${PROJECT_SOURCE_DIR}/libs/mariadb-connector-c")
set(_USE_BUNDLED_MARIADB_CONNECTOR OFF)
if(BUILD_BUNDLED_MARIADB_CONNECTOR AND EXISTS "${_BUNDLED_MARIADB_ROOT}/CMakeLists.txt")
	set(_BUNDLED_MARIADB_BUILD_DIR "${PROJECT_BINARY_DIR}/libs/mariadb-connector-c")

	# Keep the embedded connector build minimal and deterministic in this project.
	set(WITH_UNIT_TESTS OFF CACHE BOOL "Disable MariaDB Connector/C unit tests." FORCE)
	set(WITH_CURL OFF CACHE BOOL "Disable CURL support in bundled MariaDB Connector/C." FORCE)
	set(WITH_EXTERNAL_ZLIB OFF CACHE BOOL "Use bundled zlib in MariaDB Connector/C." FORCE)
	set(WITH_SSL OPENSSL CACHE STRING "TLS backend for bundled MariaDB Connector/C." FORCE)
	set(INSTALL_PLUGINDIR "lib/mariadb/plugin" CACHE PATH "Plugin install dir for bundled MariaDB Connector/C." FORCE)
	set(CLIENT_PLUGIN_DIALOG OFF CACHE STRING "Disable dialog plugin in bundled MariaDB Connector/C." FORCE)
	set(CLIENT_PLUGIN_CLIENT_ED25519 OFF CACHE STRING "Disable ed25519 plugin in bundled MariaDB Connector/C." FORCE)
	set(CLIENT_PLUGIN_CACHING_SHA2_PASSWORD OFF CACHE STRING "Disable caching_sha2 plugin in bundled MariaDB Connector/C." FORCE)
	set(CLIENT_PLUGIN_SHA256_PASSWORD OFF CACHE STRING "Disable sha256 plugin in bundled MariaDB Connector/C." FORCE)
	set(CLIENT_PLUGIN_PARSEC OFF CACHE STRING "Disable parsec plugin in bundled MariaDB Connector/C." FORCE)
	set(CLIENT_PLUGIN_MYSQL_CLEAR_PASSWORD OFF CACHE STRING "Disable mysql_clear_password plugin in bundled MariaDB Connector/C." FORCE)
	set(CLIENT_PLUGIN_ZSTD OFF CACHE STRING "Disable zstd plugin in bundled MariaDB Connector/C." FORCE)

	# Avoid network side effects from connector subproject auto-updating its own docs submodule.
	set(_PREV_GIT_EXECUTABLE "${GIT_EXECUTABLE}")
	set(GIT_EXECUTABLE "" CACHE FILEPATH "Disabled for bundled MariaDB connector subproject." FORCE)

	add_subdirectory(libs/mariadb-connector-c EXCLUDE_FROM_ALL)

	if(DEFINED _PREV_GIT_EXECUTABLE AND NOT _PREV_GIT_EXECUTABLE STREQUAL "")
		set(GIT_EXECUTABLE "${_PREV_GIT_EXECUTABLE}" CACHE FILEPATH "Path to git executable." FORCE)
	else()
		unset(GIT_EXECUTABLE CACHE)
	endif()

	if(NOT TARGET libmariadb OR NOT TARGET mariadbclient)
		message(FATAL_ERROR "Bundled MariaDB Connector/C did not produce expected targets (libmariadb, mariadbclient).")
	endif()

	set(MYSQLCAPI_FOUND TRUE)
	set(MYSQLCAPI_INCLUDE_DIR "${_BUNDLED_MARIADB_ROOT}/include;${_BUNDLED_MARIADB_BUILD_DIR}/include")
	set(MYSQLCAPI_LIBRARY "libmariadb")
	set(MYSQLCAPI_LIBRARY_SHARED "$<TARGET_FILE:libmariadb>")
	set(MYSQLCAPI_LIBRARY_STATIC "mariadbclient")
	set(_USE_BUNDLED_MARIADB_CONNECTOR ON)
	message(STATUS "Using bundled MariaDB Connector/C submodule from source: ${_BUNDLED_MARIADB_ROOT}")
endif()

if(NOT _USE_BUNDLED_MARIADB_CONNECTOR)
	find_package(MySQLCAPI REQUIRED)
endif()

set(Boost_USE_STATIC_LIBS ON)
find_package(Boost 1.57 QUIET)
if(NOT Boost_FOUND)
	set(_BUNDLED_BOOST_ROOT "${PROJECT_SOURCE_DIR}/libs/boost")
	if(EXISTS "${_BUNDLED_BOOST_ROOT}/boost/version.hpp")
		set(Boost_FOUND ON)
		set(Boost_INCLUDE_DIR "${_BUNDLED_BOOST_ROOT}")
		set(Boost_INCLUDE_DIRS "${Boost_INCLUDE_DIR}")
		message(STATUS "Using bundled Boost headers from ${_BUNDLED_BOOST_ROOT}")
	elseif(EXISTS "${_BUNDLED_BOOST_ROOT}/libs/config/include/boost/version.hpp")
		file(GLOB _BUNDLED_BOOST_INCLUDE_DIRS LIST_DIRECTORIES true "${_BUNDLED_BOOST_ROOT}/libs/*/include")
		list(REMOVE_DUPLICATES _BUNDLED_BOOST_INCLUDE_DIRS)
		if(_BUNDLED_BOOST_INCLUDE_DIRS)
			set(Boost_FOUND ON)
			set(Boost_INCLUDE_DIR "${_BUNDLED_BOOST_INCLUDE_DIRS}")
			set(Boost_INCLUDE_DIRS "${_BUNDLED_BOOST_INCLUDE_DIRS}")
			list(LENGTH _BUNDLED_BOOST_INCLUDE_DIRS _BUNDLED_BOOST_INCLUDE_DIRS_COUNT)
			message(STATUS "Using bundled Boost submodule headers from ${_BUNDLED_BOOST_ROOT} (${_BUNDLED_BOOST_INCLUDE_DIRS_COUNT} include dirs)")
		endif()
	else()
		set(BOOST_ROOT "${BOOST_ROOT}" CACHE PATH "Set the Boost root folder.")
		mark_as_advanced(CLEAR BOOST_ROOT)
		message(FATAL_ERROR "Could NOT find Boost; specify additional path. For bundled boostorg/boost, run: git -C libs/boost submodule update --init --recursive")
	endif()
endif()
mark_as_advanced(BOOST_ROOT Boost_DIR)

if(NOT TARGET fmt)
	add_subdirectory(libs/fmt)
endif()
mark_as_advanced(FMT_DOC FMT_INSTALL FMT_PEDANTIC FMT_TEST FMT_USE_CPP11)

if(EXISTS "${PROJECT_SOURCE_DIR}/libs/omp-sdk/CMakeLists.txt")
	add_subdirectory(libs/omp-sdk)
else()
	message(FATAL_ERROR "open.mp SDK submodule not found. Run: git submodule update --init --recursive")
endif()

add_subdirectory(src)
